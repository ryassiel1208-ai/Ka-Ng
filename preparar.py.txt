# Convierte el HTML original en la plantilla única (lector + administrador).
import re
s=open('plantilla/original.html',encoding='utf-8').read()
def rep(a,b,count=1):
    global s
    assert a in s, 'No encontrado: '+a[:60]
    s=s.replace(a,b,count)

s=re.sub(r'<link rel="icon" href="[^"]*">',lambda m:'<link rel="icon" href="data:image/svg+xml,'+ICON+'">',s)
rep('<title>Ka Ngäbere</title>','<title>Ka Ngäbere</title>')
rep('<meta charset="utf-8">','<meta charset="utf-8">\n<meta name="theme-color" content="#7a1f2b">')

s=re.sub(r"const DEMO=.*\n",lambda m:"const ADMIN=/*@ADMIN*/false;\nconst HIMNOS=/*@HIMNOS*/[];\nconst NA=!!window.AndroidBridge;\nlet V='home';\n",s)
rep("let H=ld('h',null)||DEMO,","let H=ADMIN?(ld('h',null)||HIMNOS):HIMNOS,")
rep("const save=()=>{sv('h',H);","const save=()=>{if(ADMIN)sv('h',H);")
rep('<button id="adm">⚙ Admin</button>',"${ADMIN?'<button id=\"adm\">⚙ Admin</button>':''}")
rep("$('#adm').onclick=admin;","if(ADMIN)$('#adm').onclick=admin;")
rep("  cur=null;edit=null;\n","  cur=null;edit=null;V='home';\n")
rep("const h=byN(n);if(!h)return home();cur=h;","const h=byN(n);if(!h)return home();cur=h;V='read';")
rep("function admin(){\n  edit=null;","function admin(){\n  edit=null;V='admin';")
rep("function form(h,isNew){\n  const old=h.n;","function form(h,isNew){\n  const old=h.n;V='form';")
rep("try{document.documentElement.requestFullscreen&&document.documentElement.requestFullscreen()}catch(e){}}",
    "if(NA)AndroidBridge.pantalla(true);else try{document.documentElement.requestFullscreen&&document.documentElement.requestFullscreen()}catch(e){}}")
rep("try{document.fullscreenElement&&document.exitFullscreen()}catch(e){}}",
    "if(NA)AndroidBridge.pantalla(false);else try{document.fullscreenElement&&document.exitFullscreen()}catch(e){}}")
old=re.search(r"  \$\('#ex'\)\.onclick=.*\n",s).group(0)

    if(NA){AndroidBridge.guardar(nm,txt)}else{const b=new Blob([txt],{type:'application/json'}),a=document.createElement('a');a.href=URL.createObjectURL(b);a.download=nm;a.click()}};
""")

rep("applyS();home();","//@ADMIN-END\nwindow.atras=()=>{\n  if($('#pres').style.display==='flex'){closeP();return true}\n  if(V==='read'){home();return true}\n  if(ADMIN&&V==='form'){admin();return true}\n  if(ADMIN&&V==='admin'){home();return true}\n  return false};\napplyS();home();")
